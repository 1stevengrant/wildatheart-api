# Beta Flag Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Wire up a single rolling `BetaFeature` Pennant flag with role + per-user-boolean membership, toggled via the Runway User resource in Statamic CP, with observer-driven cache invalidation. Branching on this flag inside DTOs/Builders is out of scope here — that lands per beta variant.

**Architecture:** A class-based Pennant feature (`App\Features\BetaFeature`) resolves true when `User::hasRole('beta-tester')` OR `users.beta_enabled === true`. The CP edits `beta_enabled` through the Runway user blueprint (super-only field). A `UserObserver::updated` hook and a `RoleAttached`/`RoleDetached` listener call `Feature::for($user)->forget(BetaFeature::class)` to keep Pennant's database cache in sync.

**Tech Stack:** Laravel 12 / PHP 8.4, Laravel Pennant 1.23 (database driver), Spatie laravel-permission, StatamicRadPack Runway, PEST 4 against `wahlocal_test`.

**Spec:** `docs/superpowers/specs/2026-04-30-beta-flag-design.md`

```php
namespace App\Features;

use App\Enums\UserRoleEnum;

class BetaFeature
```

```php
<?php

use App\Enums\UserRoleEnum;
use App\Features\BetaFeature;
use App\Models\User;
use Laravel\Pennant\Feature;
use Spatie\Permission\Models\Role;

uses(\Illuminate\Foundation\Testing\RefreshDatabase::class);

beforeEach(function () {
    Role::firstOrCreate(['name' => UserRoleEnum::BetaTester->value]);
});

it('invalidates the cached BetaFeature value when beta_enabled changes', function () {
    $user = User::factory()->create(['beta_enabled' => false]);

    // Prime the cache (resolves to false, persisted in features table).
    expect(Feature::for($user)->active(BetaFeature::class))->toBeFalse();

    // Flip the column.
    $user->update(['beta_enabled' => true]);

    // Cache should be forgotten; resolver re-runs and returns true.
    expect(Feature::for($user)->active(BetaFeature::class))->toBeTrue();
});

it('does not invalidate the cache when beta_enabled is unchanged', function () {
    $user = User::factory()->create(['beta_enabled' => false, 'name' => 'Original']);

    // Prime the cache so a features row exists.
    expect(Feature::for($user)->active(BetaFeature::class))->toBeFalse();

    expect(
        \DB::table('features')
            ->where('name', BetaFeature::class)
            ->where('scope', $user::class.'|'.$user->id)
            ->exists()
    )->toBeTrue();

    // Update an unrelated attribute.
    $user->update(['name' => 'Updated']);

    // The cached row must still exist — observer should not have called forget.
    expect(
        \DB::table('features')
            ->where('name', BetaFeature::class)
            ->where('scope', $user::class.'|'.$user->id)
            ->exists()
    )->toBeTrue();
});

it('invalidates the cache on soft-delete', function () {
    $user = User::factory()->create(['beta_enabled' => true]);

    expect(Feature::for($user)->active(BetaFeature::class))->toBeTrue();

    $user->delete();

    // The features row for this scope should have been forgotten.
    expect(
        \DB::table('features')
            ->where('name', BetaFeature::class)
            ->where('scope', $user::class.'|'.$user->id)
            ->exists()
    )->toBeFalse();
});
```

```php
use App\Features\BetaFeature;

use App\Jobs\WeGive\FindOrCreateDonorJob;
use App\Models\User;

    public function creating(User $user): void
    {
        $user->uuid = (string) Str::uuid();

    public function created(User $user): void
    {
        CheckIfUserShouldBeAdmin::dispatch($user);

        FindOrCreateDonorJob::dispatch(
            donorModel: $user,

    public function updated(User $user): void
    {
        if ($user->wasChanged('beta_enabled')) {
            $this->forgetBetaFeatureCache($user);
        }
    }

    public function deleting(User $user): void
    {
        $user->events->each->delete();
        $user->addresses()->delete();
    }

    public function deleted(User $user): void
    {
        $this->forgetBetaFeatureCache($user);
    }

    public function restored(User $user): void
    {
        //
    }

    public function forceDeleted(User $user): void
    {
        //
    }

    private function forgetBetaFeatureCache(User $user): void
    {
        try {
            Feature::for($user)->forget(BetaFeature::class);
        } catch (Throwable $e) {
            report($e);
        }
    }
}
```

```php
<?php

namespace App\Listeners;

use App\Enums\UserRoleEnum;
use App\Features\BetaFeature;
use App\Models\User;
use Laravel\Pennant\Feature;
use Spatie\Permission\Events\RoleAttached;
use Spatie\Permission\Events\RoleDetached;
use Spatie\Permission\Models\Role;
use Throwable;

class InvalidateBetaCacheOnRoleChange
{
    public function handle(RoleAttached|RoleDetached $event): void
    {
        if (! $event->model instanceof User) {
            return;
        }

        if (! $this->affectsBetaTester($event->rolesOrIds)) {
            return;
        }

        try {
            Feature::for($event->model)->forget(BetaFeature::class);
        } catch (Throwable $e) {
            report($e);
        }
    }

    private function affectsBetaTester(mixed $rolesOrIds): bool
    {
        $candidates = is_iterable($rolesOrIds) ? $rolesOrIds : [$rolesOrIds];
        $betaName = UserRoleEnum::BetaTester->value;

        foreach ($candidates as $candidate) {
            if ($candidate instanceof Role && $candidate->name === $betaName) {
                return true;
            }

            if (is_string($candidate) && $candidate === $betaName) {
                return true;
            }

            if (is_int($candidate) && Role::find($candidate)?->name === $betaName) {
                return true;
            }
        }

        return false;
    }
}
```
